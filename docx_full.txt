# Kh├┤ng cß╗Öng trß╗▒c tiß║┐p BM25_score + Cosine_similarity (sai vß╗ü mß║╖t to├ín hß╗ìc)

def hybrid_rerank(bm25_results: list, semantic_results: list) -> list:

    for rank, doc_id in enumerate(bm25_results):

    # Sß║»p xß║┐p theo tß╗òng RRF score giß║úm dß║ºn
